\title[The three-variable conflict-clause lattice]{The Three-Variable Conflict-Clause Poset: Graded Structure, Model Counting by Unobstructed Atoms, and Step-Counter Bounds, Formalized in Lean~4}
\author{Benjamin Stanley Frohman}
\email{frohmanbenjamin@gmail.com}
\date{November 4, 2026}
\begin{document}
\begin{abstract}
We study the finite poset $L_3$ of all $27$ clauses over three Boolean variables, ordered by literal inclusion (equivalently, by logical implication $c_1\models c_2$ for sub-clauses). We prove in Lean~4 with Mathlib that $L_3$ is graded with layer sizes $1,6,12,8$ and $54$ covering edges; that literal inclusion is a partial order with least element the empty clause and with a componentwise meet that is a greatest lower bound; that the alternating layer sum is $-1$; and that for every CNF formula $\Phi$ over the three variables the number of satisfying assignments equals the number of rank-$3$ clauses not implied by any clause of $\Phi$. We also formalize a step-counting state monad and prove exact tick counts and upper bounds for two explicitly defined simulation programs, including a bound of the form $(n+m)^2+2^{n/(\log n)^{1/2+\varepsilon}}$ for the block program run at block size $\lceil(\log n)^{1/2+\varepsilon}\rceil$. These bounds concern the tick counters of the defined programs only; no claim is made about the complexity of SAT, \#SAT, or the Exponential Time Hypothesis. All proofs are free of \texttt{sorry} and \texttt{native\_decide} and use only Lean's standard axioms; the main results are bundled in a single master theorem.
\end{abstract}

\section{Introduction}
Clause learning in SAT solvers manipulates clauses ordered by subsumption. For a fixed small number of variables this order is a finite poset that can be analysed exhaustively. This paper gives a complete, kernel-checked account of the case of three variables, together with an exact identity relating model counting to the poset, and a small formal framework for counting the steps of programs written in a state monad.

We emphasize at the outset what is \emph{not} claimed. The step bounds of Section~\ref{sec:steps} are statements about two specific programs whose only effect is to increment a counter by prescribed amounts (for instance, by $n^2$ and then by $2^n$). They are exact by construction and do not measure the running time of any SAT algorithm. Nothing in this paper bears on P vs.\ NP or on the Exponential Time Hypothesis.

\section{The poset $L_3$}
A \emph{variable state} is an element of $\{\bot,\top,\mathsf{F}\}$: the variable is absent, appears positively, or appears negatively. A \emph{clause} over $x_1,x_2,x_3$ (Lean: \lean{ValidClause3}) is a triple of variable states; there are $3^3=27$ of them (Lean: \lean{allValidClauses3}). The \emph{rank} of a clause is its number of literals. For variable states, $s_1\sqsubseteq s_2$ iff $s_1=\bot$ or $s_1=s_2$; for clauses, $c_1\preceq c_2$ iff this holds componentwise (Lean: \lean{ImpliesPoset}), i.e.\ the literals of $c_1$ are among those of $c_2$, so $c_1\models c_2$. A covering edge is a pair $c_1\preceq c_2$ with $\operatorname{rank}c_2=\operatorname{rank}c_1+1$. The meet $c_1\wedge c_2$ keeps, in each coordinate, a literal common to both and is $\bot$ otherwise (Lean: \lean{clauseMeet}).

\begin{theorem}[Graded structure]\label{thm:graded}
\begin{enumerate}
\item $|L_3|=27$; the layers of rank $0,1,2,3$ have sizes $1,6,12,8$, and they partition $L_3$.
\item There are exactly $54$ covering edges: $6$ from rank $0$, $24$ from rank $1$, and $24$ from rank $2$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Order and meet]\label{thm:order}
For all clauses $c,c_1,c_2,c_3$:
\begin{enumerate}
\item $c\preceq c$; if $c_1\preceq c_2$ and $c_2\preceq c_1$ then $c_1=c_2$; if $c_1\preceq c_2\preceq c_3$ then $c_1\preceq c_3$;
\item $\emptyset\preceq c$, where $\emptyset$ is the empty clause;
\item $c_1\wedge c_2\preceq c_1$ and $c_1\wedge c_2\preceq c_2$, and $c\preceq c_1\wedge c_2$ iff $c\preceq c_1$ and $c\preceq c_2$.
\end{enumerate}
\end{theorem}
The corresponding statements for single variable states (reflexivity, antisymmetry, transitivity of $\sqsubseteq$; $s\sqsubseteq s_1\wedge s_2\iff s\sqsubseteq s_1\wedge s\sqsubseteq s_2$; the meet lies below both arguments) and the componentwise characterization of $\preceq$ are proved as lemmas.

\section{Alternating sums and boundary identities}
\begin{proposition}\label{prop:euler}
$|L^{(0)}|-|L^{(1)}|+|L^{(2)}|-|L^{(3)}|=1-6+12-8=-1$ and $|L^{(1)}|-|L^{(2)}|+|L^{(3)}|=2$, where $L^{(k)}$ is the rank-$k$ layer.
\end{proposition}

For $f:L_3\to\mathbb{Z}$ and an ordered pair $(v_0,v_1)$ put $\partial f(v_0,v_1)=f(v_1)-f(v_0)$, and for a triple $(v_0,v_1,v_2)$ put $\partial^2f(v_0,v_1,v_2)=\partial f(v_1,v_2)-\partial f(v_0,v_2)+\partial f(v_0,v_1)$. Similarly, for $g$ defined on ordered pairs, put $\delta g(v_0,v_1,v_2)=g(v_1,v_2)-g(v_0,v_2)+g(v_0,v_1)$ and let $\delta^2g(v_0,\dots,v_3)$ be the alternating sum of $\delta g$ over the four faces.
\begin{proposition}\label{prop:dd}
For all $f:L_3\to\mathbb{Z}$, all $g:L_3\times L_3\to\mathbb{Z}$ and all $v_0,\dots,v_3\in L_3$: $\partial^2f(v_0,v_1,v_2)=0$ and $\delta^2g(v_0,v_1,v_2,v_3)=0$.
\end{proposition}
\begin{remark}
These are the identities $\partial\circ\partial=0$ evaluated against cochains. They hold for \emph{arbitrary} tuples of clauses, not only for chains of $L_3$, and their proofs are linear-arithmetic cancellation; they use no property of the order. The formalization does not define the order complex of $L_3$ or compute its homology.
\end{remark}

\section{Model counting by unobstructed atoms}
An assignment is $\alpha\in\{0,1\}^3$. A clause is satisfied by $\alpha$ if one of its literals is true under $\alpha$ (the empty clause is never satisfied), and a CNF formula, given as a list $\Phi$ of clauses, is satisfied if all its clauses are. Let $\#\Phi$ be the number of satisfying assignments (Lean: \lean{modelCount3}). For $\alpha$, its \emph{dual atom} $A_\alpha$ is the rank-$3$ clause with literal $\neg\alpha_i$ on each $x_i$, the unique full clause falsified by $\alpha$. A rank-$3$ clause $A$ is \emph{unobstructed} by $\Phi$ if no $c\in\Phi$ satisfies $c\preceq A$.

\begin{lemma}\label{lem:dual}
\begin{enumerate}
\item For every clause $c$ and assignment $\alpha$: $\alpha$ satisfies $c$ iff $c\not\preceq A_\alpha$.
\item Hence $\alpha$ satisfies $\Phi$ iff $A_\alpha$ is unobstructed by $\Phi$.
\item The map $\alpha\mapsto A_\alpha$, applied to the eight assignments in their standard enumeration, lists exactly the rank-$3$ layer.
\end{enumerate}
\end{lemma}
\begin{theorem}[\lean{modelCount3_eq_unobstructedAtomCount3}]\label{thm:count}
For every CNF formula $\Phi$ over $x_1,x_2,x_3$, $\#\Phi$ equals the number of rank-$3$ clauses unobstructed by $\Phi$.
\end{theorem}

\section{Step counting}\label{sec:steps}
A computation in the state monad $\mathrm{CountM}$ carries a natural-number counter; $\mathrm{tick}(k)$ adds $k$. Two programs are defined:
\begin{itemize}
\item $\mathrm{simLattice}(n)$: $\mathrm{tick}(n\cdot n)$; $\mathrm{tick}(2^n)$; return true.
\item $\mathrm{simBlock}(n,m,k)$: $\mathrm{tick}((n+m)^2)$; $\mathrm{tick}(2^{\lfloor n/\max(k,1)\rfloor})$; return true.
\end{itemize}
Let $\ell(n,\varepsilon)=(\log n)^{1/2+\varepsilon}$ (natural logarithm, real power) and $b(n,\varepsilon)=\lceil\ell(n,\varepsilon)\rceil$.

\begin{theorem}[Tick counts]\label{thm:ticks}
\begin{enumerate}
\item From counter $s$, $\mathrm{tick}(k)$ ends at $s+k$; $\mathrm{tick}(a);\mathrm{tick}(b)$ ends at $s+a+b$.
\item From $s$, $\mathrm{simLattice}(n)$ returns true with counter $s+n\cdot n+2^n$, and $\mathrm{simBlock}(n,m,k)$ returns true with counter $s+(n+m)^2+2^{\lfloor n/\max(k,1)\rfloor}$.
\item Started from $0$, $\mathrm{simLattice}(n)$ ends with exactly $n^2+2^n$ ticks (in particular at most that many), and $\mathrm{simBlock}(n,m,k)$ ends with at most $(n+m)^2+2^{\lfloor n/\max(k,1)\rfloor}$ ticks.
\end{enumerate}
\end{theorem}

\begin{theorem}[Real-logarithmic bounds]\label{thm:log}
\begin{enumerate}
\item If $n\ge2$ and $k\ge2$ are natural numbers, then $\lfloor n/k\rfloor<n$.
\item If $n\ge2$, $k\ge1$ and $\ell(n,\varepsilon)\le k$, then the final count of $\mathrm{simBlock}(n,m,k)$ from $0$ is at most $(n+m)^2+2^{n/\ell(n,\varepsilon)}$.
\item In particular, for $n\ge2$ and every real $\varepsilon$, running $\mathrm{simBlock}(n,m,b(n,\varepsilon))$ from $0$ ends with at most $(n+m)^2+2^{n/(\log n)^{1/2+\varepsilon}}$ ticks.
\item For $n\ge3$ and $\varepsilon\ge0$: $2^{n/\ell(n,\varepsilon)}<2^n$, and the final count $n^2+2^n$ of $\mathrm{simLattice}(n)$ strictly exceeds $n^2+2^{n/\ell(n,\varepsilon)}$.
\end{enumerate}
\end{theorem}
Supporting facts proved along the way: $e<3$; $\log n>0$ for $n\ge2$ and $\log n>1$ for $n\ge3$; $\ell(n,\varepsilon)>0$ for $n\ge2$ and $\ell(n,\varepsilon)>1$ for $n\ge3,\varepsilon\ge0$; $\ell\le b$; $b\ge1$ for $n\ge2$ and $b\ge2$ for $n\ge3,\varepsilon\ge0$.

\begin{proposition}[\lean{denominator_divergence}]\label{prop:div}
If $o:\mathbb{R}\to\mathbb{R}$ satisfies $o=o(1)$ as $x\to\infty$, then $(\log x)^{1/2+o(x)}\to\infty$ as $x\to\infty$.
\end{proposition}

\begin{remark}
Theorems~\ref{thm:ticks} and~\ref{thm:log} are exact statements about the counters of the two programs above, whose tick amounts are written into their definitions. They are not lower or upper bounds on any algorithm for satisfiability or model counting.
\end{remark}

\section{Master theorem}
\begin{theorem}[\lean{eth_conflict_clause_lattice_master_product_theorem}]\label{thm:master}
The conjunction of: Theorem~\ref{thm:graded} (the counts $27,1,6,12,8,54$); Theorem~\ref{thm:order}(1),(2) and the meet characterization in~(3); Proposition~\ref{prop:dd} and the first identity of Proposition~\ref{prop:euler}; Theorem~\ref{thm:count}; the bounds of Theorem~\ref{thm:ticks}(3); and Theorem~\ref{thm:log}(1),(3) and the first inequality of~(4).
\end{theorem}
